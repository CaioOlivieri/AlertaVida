46def db_path() -> Path: 

47 """Resolve o caminho do banco a cada chamada, a partir da env var. 

48 

49 `ALERTAVIDA_DB_PATH` sobrepõe o default (`data/alertavida.db` relativo ao 

50 pacote). Valor ausente ou em branco cai no default, sem levantar — mesmo 

51 padrão de `ALERTAVIDA_BUFFER_PROXIMO_GRAUS` em `domain/geographic.py`. 

52 

53 É uma função (não uma constante lida no import) de propósito: a env var 

54 pode mudar depois do import (testes, hot-reload) e cada acesso reflete o 

55 valor corrente. Não há efeito colateral de import — o `mkdir` do diretório 

56 pai foi movido para `conectar()` (ver issue #22, item A). 

57 """ 

58 valor_raw = os.getenv(ENV_DB_PATH) 

59 if valor_raw is None or not valor_raw.strip(): 

60 return DB_PATH_DEFAULT 

61 return Path(valor_raw)

64@contextlib.contextmanager 

65def conectar() -> Iterator[sqlite3.Connection]: 

66 """Abre conexão SQLite com busy_timeout para contenção multi-thread. 

67 

68 `sqlite3.Connection` usado como context manager controla apenas a 

69 *transação* (commit no sucesso / rollback na exceção) — não fecha a 

70 conexão. O `yield` abaixo acontece DENTRO de um `with conexao:`, então 

71 uma exceção levantada no bloco do chamador (`with conectar() as 

72 conexao:`) ainda dispara rollback (issue #40 — invariante da outbox 

73 transacional, ver wiki/patterns/resilience-invariants.md #4) antes do 

74 `finally` fechar a conexão de fato. 

75 

76 O diretório pai do banco é criado aqui (não no import — issue #22 item A): 

77 importar o módulo é livre de efeito colateral no filesystem. 

78 """ 

79 caminho = db_path() 

80 caminho.parent.mkdir(parents=True, exist_ok=True) 

81 conexao = sqlite3.connect(caminho) 

82 conexao.execute("PRAGMA busy_timeout=5000") 

83 # foreign_keys é por-conexão e vira no-op dentro de uma transação — tem de 

84 # ser ligado ANTES do `with conexao:` abaixo (issue #22, item B). 

85 conexao.execute("PRAGMA foreign_keys=ON") 

86 try: 

87 with conexao: 

88 yield conexao 

89 finally: 

90 conexao.close()

93class SchemaIncompativelError(Exception): 

94 """Banco existente tem schema incompatível com a versão atual do código. 

95 

96 Dois gatilhos, ambos sem caminho de migration automática (SQLite não 

97 suporta a mudança via ALTER TABLE): 

98 - tabela `alertas` pré-A.1 (sem coluna `id` ou `fonte`) — a Camada 4 Parte 

99 A.1 (09/05/2026) trocou PK composta por surrogate key; 

100 - tabela `eventos` sem a FOREIGN KEY para `alertas.id` — banco criado antes 

101 da issue #22 (2026-07-28), que passou a exigir integridade referencial. 

102 """

105def _verificar_compatibilidade_schema(conexao: sqlite3.Connection) -> None: 

106 """Verifica se o schema existente é compatível com a versão atual. 

107 

108 Casos: 

109 - Tabela `alertas` não existe -> OK, criar_banco() vai criá-la 

110 - Tabela `alertas` existe com `id` + `fonte` -> OK, _migrar_banco() aditiva 

111 - Tabela `alertas` existe sem `id` ou sem `fonte` -> SchemaIncompativelError 

112 - Tabela `eventos` não existe -> OK, criar_banco() vai criá-la (com FK) 

113 - Tabela `eventos` existe SEM FOREIGN KEY -> SchemaIncompativelError (#22) 

114 

115 Pré-condição: chamada como primeira operação dentro de criar_banco(), 

116 antes de qualquer CREATE TABLE ou ALTER TABLE. 

117 """ 

118 colunas_alertas = {row[1] for row in conexao.execute("PRAGMA table_info(alertas)")} 

119 if colunas_alertas: 

120 faltantes = {"id", "fonte"} - colunas_alertas 

121 if faltantes: 

122 raise SchemaIncompativelError( 

123 f"Schema do banco em '{db_path()}' é incompatível com a versão atual.\n" 

124 f"\n" 

125 f"Detectado: tabela `alertas` sem coluna(s): {sorted(faltantes)}.\n" 

126 f"Provável origem: banco criado antes da Camada 4 Parte A.1 (09/05/2026).\n" 

127 f"\n" 

128 f"A Camada 4 Parte A.1 introduziu ruptura estrutural (surrogate key + " 

129 f"UNIQUE composto) sem caminho de migration automática — bancos pré-A.1 " 

130 f"precisam ser recriados.\n" 

131 f"\n" 

132 f"Ação: apague o arquivo do banco e rode criar_banco() novamente.\n" 

133 f"Se houver dados a preservar, exporte para JSON antes de apagar." 

134 ) 

135 

136 # Item B da issue #22: `eventos` tem de declarar a FK para `alertas.id`. 

137 # SQLite não adiciona FK via ALTER TABLE, e recriar a tabela está fora da 

138 # política aditiva de _migrar_banco — logo, banco pré-#22 é barrado aqui. 

139 colunas_eventos = {row[1] for row in conexao.execute("PRAGMA table_info(eventos)")} 

140 if colunas_eventos: 

141 tem_fk = bool(conexao.execute("PRAGMA foreign_key_list(eventos)").fetchall()) 

142 if not tem_fk: 

143 raise SchemaIncompativelError( 

144 f"Schema do banco em '{db_path()}' é incompatível com a versão atual.\n" 

145 f"\n" 

146 f"Detectado: tabela `eventos` sem FOREIGN KEY para `alertas.id`.\n" 

147 f"Provável origem: banco criado antes da issue #22 (2026-07-28).\n" 

148 f"\n" 

149 f"A issue #22 passou a exigir integridade referencial em `eventos`. " 

150 f"SQLite não adiciona FOREIGN KEY via ALTER TABLE, e recriar a tabela " 

151 f"está fora da política aditiva de _migrar_banco() — bancos pré-#22 " 

152 f"precisam ser recriados.\n" 

153 f"\n" 

154 f"Ação: apague o arquivo do banco e rode criar_banco() novamente.\n" 

155 f"Se houver dados a preservar, exporte para JSON antes de apagar." 

156 )

159def _migrar_banco(conexao: sqlite3.Connection) -> None: 

160 """Aplica migrations aditivas/de limpeza idempotentes ao schema existente. 

161 

162 Cobre as colunas COBRADE da Camada 4 Parte A.2 (`cobrade_codigo`, 

163 `fonte_classificacao`), adicionadas via `ALTER TABLE` apenas quando 

164 ausentes; a remoção de `assinatura` (issue #8 B1 — coluna nunca lida 

165 nem escrita com valor real, resquício da abordagem de hash pré- 

166 `ult_atualizacao`); a remoção dos índices especulativos `idx_uf`, 

167 `idx_evento`, `idx_nivel` (issue #11 D3 — sem query real hoje, custo 

168 de escrita sem benefício até a Camada 6 definir os filtros reais); e 

169 a coluna aditiva `descricao` (issue #11 D4 — write-only no domínio 

170 até aqui, `NasaEonetSource` já populava com o título do evento mas o 

171 dado morria na ingestão). A migration de PK composta 

172 para surrogate (A.1.4) nunca passou por aqui — o banco estava vazio no 

173 refator e rupturas estruturais são barradas antes por 

174 `_verificar_compatibilidade_schema`. É o ponto de extensão obrigatório 

175 para qualquer mudança de schema aditiva ou de limpeza futura. 

176 """ 

177 # Camada 4 / A.2 — colunas COBRADE 

178 cursor = conexao.execute("PRAGMA table_info(alertas)") 

179 colunas_existentes = {row[1] for row in cursor.fetchall()} 

180 

181 if "cobrade_codigo" not in colunas_existentes: 

182 conexao.execute("ALTER TABLE alertas ADD COLUMN cobrade_codigo TEXT NULL") 

183 

184 if "fonte_classificacao" not in colunas_existentes: 

185 conexao.execute( 

186 "ALTER TABLE alertas ADD COLUMN fonte_classificacao " 

187 "TEXT NOT NULL DEFAULT 'INDETERMINADA'" 

188 ) 

189 

190 # Manutenibilidade #8 B1 — assinatura nunca teve valor real (sempre 

191 # inserida como NULL literal); DROP seguro, sem dado a preservar. 

192 if "assinatura" in colunas_existentes: 

193 conexao.execute("ALTER TABLE alertas DROP COLUMN assinatura") 

194 

195 # Manutenibilidade #11 D3 — idx_uf/idx_evento/idx_nivel eram especulativos 

196 # (nenhuma query atual filtra por essas colunas; Camada 6 ainda não 

197 # existe). DROP INDEX não afeta dados, apenas remove custo de escrita. 

198 conexao.execute("DROP INDEX IF EXISTS idx_uf") 

199 conexao.execute("DROP INDEX IF EXISTS idx_evento") 

200 conexao.execute("DROP INDEX IF EXISTS idx_nivel") 

201 

202 # Manutenibilidade #11 D4 — descricao era write-only no domínio (NasaEonetSource 

203 # já a populava com o título do evento, mas o dado morria na ingestão). 

204 if "descricao" not in colunas_existentes: 

205 conexao.execute("ALTER TABLE alertas ADD COLUMN descricao TEXT NULL") 

206 

207 # Issue #59 — eventos de ciclo de vida de Incidente não podem reusar 

208 # agregado_id (FK fixa para alertas(id) desde a #22). Coluna aditiva 

209 # NULL com sua própria FK; SQLite aceita REFERENCES em ADD COLUMN 

210 # (verificado empiricamente — só não aceita para colunas NOT NULL sem 

211 # default constante). `incidentes` já existe neste ponto: é criada em 

212 # criar_banco() antes desta chamada. Ver [[decisions/agregado-incidente-id]]. 

213 cursor_eventos = conexao.execute("PRAGMA table_info(eventos)") 

214 colunas_eventos = {row[1] for row in cursor_eventos.fetchall()} 

215 if "agregado_incidente_id" not in colunas_eventos: 

216 conexao.execute( 

217 "ALTER TABLE eventos ADD COLUMN agregado_incidente_id " 

218 "INTEGER NULL REFERENCES incidentes(id)" 

219 ) 

220 

221 # Issue #60 — backfill do índice espacial (R-Tree) para bancos que já 

222 # tinham alertas antes desta migration. Sem isso, alertas antigos 

223 # nunca apareceriam como candidatos de correlação (o INSERT normal do 

224 # índice só cobre alertas CRIADOs a partir de agora, em 

225 # aplicar_resultado_deteccao). Idempotente via NOT IN; `idx_alertas_espacial` 

226 # já existe neste ponto (criada em criar_banco() antes desta chamada). 

227 conexao.execute( 

228 """ 

229 INSERT INTO idx_alertas_espacial (id, min_lat, max_lat, min_lon, max_lon) 

230 SELECT id, latitude, latitude, longitude, longitude 

231 FROM alertas 

232 WHERE id NOT IN (SELECT id FROM idx_alertas_espacial) 

233 """ 

234 )

238 with conectar() as conexao: 

239 conexao.execute("PRAGMA journal_mode=WAL") 

240 _verificar_compatibilidade_schema(conexao) # detecta bancos pré-A.1 

241 conexao.execute( 

242 """ 

243 CREATE TABLE IF NOT EXISTS alertas ( 

244 id INTEGER PRIMARY KEY AUTOINCREMENT, 

245 fonte TEXT NOT NULL DEFAULT 'CEMADEN', 

246 cod_alerta TEXT NOT NULL, 

247 municipio TEXT, 

248 uf TEXT, 

249 evento TEXT, 

250 nivel TEXT, 

251 datahoracriacao TEXT, 

252 detectado_em TEXT NOT NULL, 

253 codibge INTEGER, 

254 latitude REAL NOT NULL, 

255 longitude REAL NOT NULL, 

256 escopo_geografico TEXT NOT NULL DEFAULT 'INDETERMINADO', 

257 ult_atualizacao TEXT, 

258 status_interno TEXT NOT NULL DEFAULT 'ATIVO', 

259 visto_ultima_vez TEXT NOT NULL DEFAULT '', 

260 rodadas_ausente INTEGER NOT NULL DEFAULT 0, 

261 cobrade_codigo TEXT NULL, 

262 fonte_classificacao TEXT NOT NULL DEFAULT 'INDETERMINADA', 

263 descricao TEXT NULL, 

264 UNIQUE (fonte, cod_alerta) 

265 ) 

266 """ 

267 )

269 conexao.execute( 

270 "CREATE INDEX IF NOT EXISTS idx_escopo_geografico ON alertas (escopo_geografico)"

273 """

286 ) 

287 """

293 """

407def buscar_snapshots(fonte: FonteDado) -> list[AlertaSnapshot]: 

408 """Retorna snapshots de todos os alertas (qualquer status) da fonte. 

409 

410 Cada snapshot carrega sua `fonte` como campo (lido da row, não 

411 hardcoded do parâmetro) — robustez contra futuras mudanças no 

412 WHERE da query. Removeu filtro `status_interno = 'ATIVO'` para que 

413 alertas RESOLVIDO que reaparecem sejam detectados corretamente. 

414 """ 

415 with conectar() as conexao: 

416 cursor = conexao.execute( 

417 """ 

418 SELECT cod_alerta, fonte, ult_atualizacao, 

419 rodadas_ausente, status_interno 

420 FROM alertas 

421 WHERE fonte = ? 

422 """, 

423 (fonte.value,), 

424 ) 

425 return [ 

426 AlertaSnapshot( 

427 cod_alerta=row[0], 

428 fonte=FonteDado.from_string(row[1]), 

429 ult_atualizacao=row[2], 

430 rodadas_ausente=row[3], 

431 status_interno=row[4], 

432 ) 

433 for row in cursor.fetchall() 

434 ]

437def _executar_retornando_id(conexao: sqlite3.Connection, sql: str, params: tuple) -> int | None: 

438 """Executa um UPDATE ... RETURNING id e devolve o id (None se 0 linhas). 

439 

440 Substitui o par UPDATE + SELECT id por uma única query. RETURNING exige 

441 SQLite >= 3.35 (Python 3.13 embute >= 3.40). 

442 """ 

443 row = conexao.execute(sql, params).fetchone() 

444 return row[0] if row is not None else None

447def aplicar_resultado_deteccao( 

448 resultado: ResultadoDeteccao, 

449 alertas_por_codigo: dict[str, "Alerta"], 

450 agora: str, 

451) -> dict[str, int]: 

452 """Persiste o resultado do ChangeDetector atomicamente. 

453 

454 INSERT/UPDATE em alertas e INSERT em eventos ocorrem na mesma transação 

455 SQLite — outbox pattern. `agregado_id` em eventos referencia o `id` 

456 surrogate da tabela `alertas`. 

457 

458 A fonte de cada código é obtida via `resultado.fonte_por_codigo[cod]` 

459 (populado pelo detector) — não recebe `fonte` como parâmetro. Isso 

460 permite que rodadas multi-fonte (Camada 5+) sejam tratadas sem 

461 mudanças nesta função. 

462 

463 Retorna `{cod_alerta: alerta_id}` para todo evento com um `agregado_id` 

464 resolvido (CRIADO/ATUALIZADO/REATIVADO/RESOLVIDO) — Tell, Don't Ask 

465 (mesmo princípio de `ResultadoDeteccao`): esta função já calcula o id 

466 surrogate de cada alerta para o INSERT/UPDATE de `alertas`, e a 

467 integração de correlação (#61) precisa exatamente desses ids logo em 

468 seguida para CRIADO/REATIVADO/RESOLVIDO — reconsultar por 

469 (fonte, cod_alerta) seria uma query redundante por alerta. 

470 """ 

471 ids_por_codigo: dict[str, int] = {} 

472 with conectar() as conexao: 

473 for evento in resultado.eventos: 

474 agregado_id: int | None = None 

475 

476 if evento.tipo is TipoEventoDetectado.CRIADO: 

477 alerta = alertas_por_codigo[evento.cod_alerta] 

478 ult = alerta.ult_atualizacao.isoformat() if alerta.ult_atualizacao else None 

479 cursor = conexao.execute( 

480 """ 

481 INSERT INTO alertas ( 

482 fonte, cod_alerta, municipio, uf, evento, nivel, 

483 datahoracriacao, detectado_em, codibge, 

484 latitude, longitude, escopo_geografico, ult_atualizacao, 

485 status_interno, visto_ultima_vez, rodadas_ausente, 

486 cobrade_codigo, fonte_classificacao, descricao 

487 ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 'ATIVO', ?, 0, ?, ?, ?) 

488 """, 

489 ( 

490 alerta.fonte.value, 

491 alerta.cod_alerta, 

492 alerta.municipio.nome if alerta.municipio is not None else None, 

493 alerta.municipio.uf if alerta.municipio is not None else None, 

494 alerta.tipo_evento.value,

858def buscar_incidente_atual(alerta_id: int) -> int | None: 

859 """Segue o redirect de fusão (`fundido_em`) até o Incidente sobrevivente 

860 ao qual `alerta_id` pertence hoje. 

861 

862 `incidente_membros` nunca move linhas entre incidentes (Round 1, Q6 — 

863 fusão é redirect, nunca realocação), então a associação original de um 

864 alerta pode apontar para um Incidente já fundido em outro; este helper 

865 resolve a cadeia até o id atual. Retorna `None` se o alerta nunca foi 

866 correlacionado (nenhuma linha em `incidente_membros`) — caso de um 

867 alerta ainda não processado por `avaliar_candidatos_correlacao`/a ação 

868 de #61 sobre ela. 

869 """ 

870 with conectar() as conexao: 

871 row = conexao.execute( 

872 "SELECT incidente_id FROM incidente_membros WHERE alerta_id = ?", 

873 (alerta_id,), 

874 ).fetchone() 

875 if row is None: 

876 return None 

877 incidente_id: int = row[0] 

878 while True: 

879 redirecionamento = conexao.execute( 

880 "SELECT fundido_em FROM incidentes WHERE id = ?", (incidente_id,) 

881 ).fetchone() 

882 if redirecionamento is None or redirecionamento[0] is None: 

883 return incidente_id 

884 incidente_id = redirecionamento[0]

902def todos_membros_resolvidos(incidente_id: int) -> bool: 

903 """True quando TODO Alerta membro do Incidente — e de qualquer Incidente 

904 fundido nele, transitivamente — está com `status_interno = 'RESOLVIDO'` 

905 (Round 1, Q5: resolve só quando o ÚLTIMO membro não-resolvido resolve). 

906 

907 A árvore de fusão importa porque `fundir_incidentes` nunca move linhas 

908 de `incidente_membros` para o sobrevivente (redirect append-only) — os 

909 membros do Incidente fundido continuam com `incidente_id` apontando 

910 para ele, não para o sobrevivente, então contar só os membros do id 

911 passado ignoraria membros herdados por fusão.

916 WITH RECURSIVE arvore_fusao(id) AS ( 

917 SELECT ? 

918 UNION ALL 

919 SELECT i.id FROM incidentes i 

920 JOIN arvore_fusao f ON i.fundido_em = f.id 

921 ) 

922 SELECT COUNT(*) 

923 FROM incidente_membros im 

924 JOIN alertas a ON a.id = im.alerta_id 

925 JOIN arvore_fusao f ON f.id = im.incidente_id 

926 WHERE a.status_interno != 'RESOLVIDO' 

927 """, 

928 (incidente_id,), 

929 ).fetchone() 

930 return row[0] == 0

934# Blocking de correlação (Camada 5, issue #60) — geração de candidatos e 

935# instrumentação. Decidir se um par (Alerta, Incidente candidato) é o mesmo 

936# evento é responsabilidade do núcleo puro em domain/correlacao.py (#58); 

937# este módulo só gera candidatos baratos (bbox + janela) e avalia cada um 

938# através daquele núcleo, registrando TODO par avaliado em 

939# correlacao_observacoes — inclusive NAO_VINCULA (Round 1, Q1, "instrument 

940# before calibrating"; wiki/projects/layer-5-correlation.md). 

941# 

942# Candidatos = SÓ incidentes com status ATIVO, fundido_em IS NULL, cujo 

943# bbox intersecte a posição (com buffer) do alerta avaliado e cuja janela de 

944# tempo por tipo ainda esteja aberta (Round 1, Q6 — custo O(incidentes 

945# abertos), nunca O(histórico)): a query só toca incidentes já filtrados por 

946# `idx_incidentes_status_fundido` e os membros DESSES incidentes via 

947# `idx_alertas_espacial`, nunca a tabela `alertas` inteira. 

948# --------------------------------------------------------------------------- 

949 

950# Buffer de blocking em graus decimais — NUNCA usado na decisão (#58 decide 

951# com haversine exata sobre coordenadas reais); só alimenta o WHERE do 

952# R-Tree para superselecionar candidatos (Round 1, Q2: bbox pode 

953# superselecionar, nunca subselecionar). Convertido de DISTANCIA_MAXIMA_KM 

954# (#58) usando a pior compressão de longitude do território brasileiro 

955# (~34°S, 1° de longitude ≈ 92 km — valor conservador, arredondado para 

956# baixo). Na latitude, 111 km/grau é aproximadamente constante em todo o 

957# país, então o mesmo buffer também cobre esse eixo com folga. 

958# 

959# GARANTIA VÁLIDA SÓ PARA LATITUDES BRASILEIRAS (até ~34°S). NasaEonetSource 

960# ingere eventos globais sem filtro de bbox (decision record, "Global NASA 

961# EONET ingestion"), e acima de |34°| a compressão de longitude piora (a 

962# 60°S, 1° de longitude ≈ 55 km — o mesmo buffer cobre só ~30 km ali), 

963# então FORA do Brasil o blocking pode SUBSELECIONAR candidatos por 

964# longitude, violando a garantia acima. Limitação conhecida e aceita — o 

965# produto é brasileiro e o viés do projeto é para separar (Round 2), não 

966# para vincular de mais —, registrada aqui para não ficar silenciosa. 

967_KM_POR_GRAU_LONGITUDE_MINIMO_BRASIL: Final[float] = 92.0 

968BUFFER_BLOQUEIO_GRAUS: Final[float] = DISTANCIA_MAXIMA_KM / _KM_POR_GRAU_LONGITUDE_MINIMO_BRASIL

972# entrada é (janela_antes, janela_depois) em segundos: quanto o onset do 

973# alerta avaliado pode preceder (antes) ou suceder (depois) o onset do

978# código, quando calibrar contra pares confirmados reais (latência de 

979# detecção difere por fonte, então antes/depois não devem convergir para o

982JANELA_ABERTA_SEGUNDOS_POR_TIPO: Final[dict[TipoEvento, tuple[float, float]]] = {

1009 if not datahoracriacao: 

1010 raise ValueError( 

1011 f"Alerta {fonte}/{cod_alerta} sem datahoracriacao — não é possível " 

1012 "correlacionar sem um momento de onset."

1015 fonte=FonteDado.from_string(fonte), 

1016 cod_alerta=cod_alerta, 

1017 tipo_evento=TipoEvento.from_string(evento), 

1018 cobrade_codigo=cobrade_codigo, 

1019 codigo_ibge=codibge, 

1020 latitude=latitude,

1026def _buscar_incidentes_candidatos( 

1027 conexao: sqlite3.Connection, 

1028 candidato_alerta: CandidatoCorrelacao, 

1029 janela_antes_segundos: float, 

1030 janela_depois_segundos: float, 

1031) -> list[tuple[int, CandidatoCorrelacao]]: 

1032 """Blocking: incidentes abertos com um membro dentro do bbox bufferizado 

1033 E dentro da janela de tempo por tipo, um por incidente. 

1034 

1035 Estágio 1 (bbox, R-Tree) roda inteiramente em SQL — é o que realmente 

1036 precisa de índice espacial. Estágio 2 (janela de tempo) roda em Python 

1037 sobre o resultado já filtrado pelo bbox (pequeno, O(incidentes abertos)), 

1038 evitando aritmética de data em SQL sobre strings ISO. Quando um 

1039 incidente tem mais de um membro dentro do bbox, o membro mais 

1040 recentemente associado (`incidente_membros.criado_em` mais alto) 

1041 representa o incidente na comparação — decisão v1, documentada em 

1042 wiki/decisions/incidente-representante-blocking.md. 

1043 """ 

1044 linhas = conexao.execute( 

1045 """ 

1046 SELECT im.incidente_id, im.criado_em, 

1047 a.fonte, a.cod_alerta, a.evento, a.cobrade_codigo, a.codibge, 

1048 a.latitude, a.longitude, a.datahoracriacao 

1049 FROM idx_alertas_espacial r 

1050 JOIN incidente_membros im ON im.alerta_id = r.id 

1051 JOIN incidentes i ON i.id = im.incidente_id 

1052 JOIN alertas a ON a.id = r.id 

1053 WHERE i.status = 'ATIVO' AND i.fundido_em IS NULL 

1054 AND r.min_lat <= ? AND r.max_lat >= ? 

1055 AND r.min_lon <= ? AND r.max_lon >= ? 

1056 """, 

1057 ( 

1058 candidato_alerta.latitude + BUFFER_BLOQUEIO_GRAUS, 

1059 candidato_alerta.latitude - BUFFER_BLOQUEIO_GRAUS, 

1060 candidato_alerta.longitude + BUFFER_BLOQUEIO_GRAUS, 

1061 candidato_alerta.longitude - BUFFER_BLOQUEIO_GRAUS, 

1062 ), 

1063 ).fetchall() 

1064 

1065 representante_por_incidente: dict[int, tuple[str, tuple]] = {} 

1066 for incidente_id, criado_em_membro, *campos_alerta in linhas: 

1067 if not campos_alerta[-1]: # datahoracriacao ausente — membro inutilizável 

1068 continue 

1069 atual = representante_por_incidente.get(incidente_id) 

1070 if atual is None or criado_em_membro > atual[0]: 

1071 representante_por_incidente[incidente_id] = (criado_em_membro, tuple(campos_alerta)) 

1072 

1073 candidatos: list[tuple[int, CandidatoCorrelacao]] = [] 

1074 for incidente_id, (_, campos_alerta) in representante_por_incidente.items(): 

1075 representante = _construir_candidato_correlacao(campos_alerta) 

1076 # Sinal importa (Round 1, Q1 — janela assimétrica): delta_t > 0 

1077 # significa que o alerta avaliado tem onset DEPOIS do representante; 

1078 # delta_t < 0, ANTES. Cada lado é checado contra sua própria janela. 

1079 delta_t = (candidato_alerta.momento_onset - representante.momento_onset).total_seconds() 

1080 if -janela_antes_segundos <= delta_t <= janela_depois_segundos: 

1081 candidatos.append((incidente_id, representante)) 

1082 return candidatos

1115def avaliar_candidatos_correlacao(alerta_id: int, agora: str) -> list[ObservacaoCandidato]: 

1116 """Gera candidatos (blocking), avalia cada um via `decidir_correlacao` 

1117 (#58) e grava uma linha em `correlacao_observacoes` por par avaliado — 

1118 inclusive quando não há nenhum candidato. Não muta `incidentes`/ 

1119 `incidente_membros`: agir sobre VINCULA/REVISAO (criar, juntar, fundir) 

1120 é responsabilidade de quem chama (#61). 

1121 """ 

1122 with conectar() as conexao: 

1123 row_alerta = conexao.execute( 

1124 """ 

1125 SELECT fonte, cod_alerta, evento, cobrade_codigo, codibge, 

1126 latitude, longitude, datahoracriacao 

1127 FROM alertas WHERE id = ? 

1128 """, 

1129 (alerta_id,), 

1130 ).fetchone() 

1131 if row_alerta is None: 

1132 raise ValueError(f"Alerta id={alerta_id} não encontrado") 

1133 candidato_alerta = _construir_candidato_correlacao(row_alerta) 

1134 

1135 janela_antes, janela_depois = JANELA_ABERTA_SEGUNDOS_POR_TIPO[candidato_alerta.tipo_evento] 

1136 candidatos = _buscar_incidentes_candidatos( 

1137 conexao, candidato_alerta, janela_antes, janela_depois 

1138 ) 

1139 

1140 resultados: list[ObservacaoCandidato] = [] 

1141 if not candidatos: 

1142 decisao_vazia = DecisaoCorrelacao( 

1143 resultado=ResultadoDecisao.NAO_VINCULA, 

1144 score=0.0, 

1145 motivo="sem_candidatos", 

1146 distancia_km=0.0, 

1147 delta_t_segundos=0.0, 

1148 ) 

1149 _inserir_observacao_correlacao( 

1150 conexao,

1152 incidente_id=None,

1156 ) 

1157 resultados.append(ObservacaoCandidato(incidente_id=None, decisao=decisao_vazia)) 

1158 else: 

1159 for incidente_id, representante in candidatos:

1162 candidato_alerta.codigo_ibge is not None 

1163 and candidato_alerta.codigo_ibge == representante.codigo_ibge 

1164 )
