1"""Gera o site estático do acervo a partir de uma exportação (issue #158). 

2 

3Lê só os arquivos de `alertavida.acervo.exportar` — nunca o banco — e escreve 

4HTML + um CSS, sem JavaScript, com links relativos (abre por `file://` e 

5funciona em qualquer subcaminho de hospedagem): 

6 

7- `index.html` — o que está ativo agora, a faixa dos últimos 30 dias (alertas 

8 por dia e quanto do dia o coletor esteve ligado) e os municípios por UF; 

9- `municipio/<codibge>.html` — o histórico de cada município com alerta; 

10- `dados.html` + `dados/` — os CSVs, o dicionário, os metadados e a 

11 procedência técnica (nome e SHA-256 do backup). 

12 

13Desenho para o celular primeiro: alertas são uma lista semântica (rótulo e 

14valor em duas colunas), nunca uma tabela que role de lado. Nível de 

15risco nas cores convencionais da Defesa Civil, sempre com o texto ao lado. 

16Fontes do sistema (nenhum pedido a terceiros). Todo texto vindo das fontes 

17passa por `html.escape`. Horários na tela em horário de Brasília (UTC−3, sem 

18horário de verão desde 2019); os CSVs ficam em UTC. Determinístico: a mesma 

19exportação gera os mesmos bytes. 

20 

21Uso: 

22 

23 uv run python -m alertavida.acervo.site \\ 

24 --export /var/lib/alertavida/export --saida /var/lib/alertavida/site

358def _faixa(alertas: list[AlertaSite], cobertura: list[dict[str, str]], contexto: Contexto) -> str: 

359 fim = _instante(contexto.atualizado_utc).astimezone(BRASILIA).date() 

360 dias = [fim - timedelta(days=d) for d in range(DIAS_RECENTES - 1, -1, -1)] 

361 por_dia: dict[date, int] = defaultdict(int) 

362 for a in alertas: 

363 por_dia[_instante(a.inicio_utc).astimezone(BRASILIA).date()] += 1 

364 # A cobertura é medida por dia em UTC; para a faixa basta o dia correspondente. 

365 cob = {c["dia"]: float(c["cobertura"]) for c in cobertura if c["fonte"] == "CEMADEN"} 

366 maximo = max([1, *(por_dia.get(d, 0) for d in dias)]) 

367 

368 itens = [] 

369 for posicao, dia in enumerate(dias): 

370 n = por_dia.get(dia, 0) 

371 fracao = cob.get(dia.isoformat()) 

372 if fracao is None: 

373 coleta = '<span class="coleta sem-registro"></span>' 

374 texto_coleta = "sem registro de coleta" 

375 else: 

376 coleta = f'<span class="coleta" style="--cobertura:{max(fracao, 0.08):.2f}"></span>' 

377 texto_coleta = f"coletor ligado em {fracao:.0%} do dia" 

378 mostrar = posicao % 7 == (DIAS_RECENTES - 1) % 7 

379 marca = f"{dia:%d/%m}" if mostrar else "" 

380 alertas_txt = "1 alerta" if n == 1 else f"{n} alertas" 

381 itens.append( 

382 f'<li style="--alertas:{n}"><span class="barra"></span>{coleta}' 

383 f'<span class="marca-dia" aria-hidden="true">{marca}</span>' 

384 f'<span class="sr">{dia:%d/%m}: {alertas_txt}; {texto_coleta}</span></li>' 

385 ) 

386 return ( 

387 f'<ol class="faixa" style="--max:{maximo}">{"".join(itens)}</ol>' 

388 '<p class="legenda nota"><span class="l-alertas">alertas iniciados no dia</span>' 

389 '<span class="l-coleta">coletor ligado (mais claro: menos horas)</span>' 

390 '<span class="l-sem">sem registro de coleta</span></p>' 

391 )

395 alertas: list[AlertaSite], 

396 avisos_por_alerta: dict[str, list[AvisoSite]], 

397 cobertura: list[dict[str, str]], 

398 contexto: Contexto, 

399) -> str: 

400 ativos = sorted(

406 if ativos: 

407 agora = (

409 f"</strong> em {municipios_ativos} "

412 ) 

413 else: 

414 agora = f"Nenhum alerta ativo em {em_brasilia(contexto.atualizado_utc)}." 

415 partes = [ 

416 "<h1>Alertas de risco do CEMADEN, com os avisos do INMET</h1>",

420 "alerta começou. Horários de Brasília.</p>", 

421 _AVISO_OFICIAL, 

422 f'<h2 id="t-faixa">Últimos {DIAS_RECENTES} dias</h2>',

426 '<h2 id="t-ativos">Ativos agora</h2>', 

427 ] 

428 if ativos: 

429 partes.append('<ul class="alertas">') 

430 partes += [ 

431 _item_alerta( 

432 a, 

433 avisos_por_alerta.get(a.cod_alerta, []), 

434 contexto, 

435 lugar=_link_municipio(a), 

436 com_situacao=False, 

437 ) 

438 for a in ativos 

439 ] 

440 partes.append("</ul>") 

441 else: 

442 partes.append("<p>Nenhum alerta ativo no momento.</p>") 

443 

444 por_uf: dict[str, dict[str, tuple[str, int]]] = defaultdict(dict) 

445 for a in alertas: 

446 nome, total = por_uf[a.uf].get(a.codibge, (a.municipio, 0)) 

447 por_uf[a.uf][a.codibge] = (nome, total + 1) 

448 total_municipios = sum(len(m) for m in por_uf.values()) 

449 partes.append(f'<h2 id="t-municipios">Municípios com alertas ({total_municipios})</h2>') 

450 for uf in sorted(por_uf): 

451 itens = sorted(por_uf[uf].items(), key=lambda item: (item[1][0], item[0])) 

452 n = len(itens) 

453 partes.append( 

454 f'<details><summary>{escape(uf)} <span class="contagem">' 

455 f"{n} {'município' if n == 1 else 'municípios'}</span></summary>" 

456 '<ul class="municipios">' 

457 ) 

458 partes += [ 

459 f'<li><a href="municipio/{escape(cod)}.html">{escape(nome_proprio(nome))}</a> ' 

460 f'<span class="sem">({total})</span></li>' 

461 for cod, (nome, total) in itens 

462 ] 

463 partes.append("</ul></details>") 

464 return _pagina("AlertaVida: alertas de risco do CEMADEN", "\n".join(partes), "", contexto)

468 alertas: list[AlertaSite], avisos_por_alerta: dict[str, list[AvisoSite]], contexto: Contexto 

469) -> str: 

470 alertas = sorted(alertas, key=lambda a: (a.inicio_utc, a.cod_alerta), reverse=True) 

471 mais_antigo = alertas[-1] 

472 nome = nome_proprio(mais_antigo.municipio) 

473 ativos = sum(1 for a in alertas if a.status == "ATIVO") 

474 total = "1 alerta" if len(alertas) == 1 else f"{len(alertas)} alertas" 

475 resumo = f"{total} do CEMADEN desde {em_brasilia(mais_antigo.inicio_utc)[:10]}" + ( 

476 f"; {ativos} {'ativo' if ativos == 1 else 'ativos'} agora." if ativos else "." 

477 ) 

478 partes = [ 

479 f'<h1>{escape(nome)} <span class="uf">{escape(mais_antigo.uf)}</span></h1>', 

480 f'<p class="agora">{resumo}</p>', 

481 _AVISO_OFICIAL, 

482 '<h2 id="t-historico">Histórico</h2>', 

483 ] 

484 mes_atual = None 

485 for a in alertas: 

486 mes = _mes_por_extenso(a.inicio_utc) 

487 if mes != mes_atual: 

488 if mes_atual is not None: 

489 partes.append("</ol>") 

490 partes += [f"<h3>{mes}</h3>", '<ol class="alertas">'] 

491 mes_atual = mes 

492 partes.append( 

493 _item_alerta( 

494 a, avisos_por_alerta.get(a.cod_alerta, []), contexto, lugar=None, com_situacao=True 

495 ) 

496 ) 

497 partes.append("</ol>") 

498 partes.append( 

499 '<p class="nota">"Saiu do painel" é quando o AlertaVida percebeu que o alerta deixou ' 

500 "de aparecer no painel do CEMADEN. Se o coletor estava desligado, ele pode ter saído " 

501 f"antes. Código IBGE do município: {escape(mais_antigo.codibge)}. " 

502 'Os dados completos estão em <a href="../dados.html">Dados</a>.</p>' 

503 ) 

504 return _pagina( 

505 f"{nome} ({mais_antigo.uf}): alertas — AlertaVida", "\n".join(partes), "../", contexto 

506 )

538def gerar_site(export: Path, saida: Path) -> int: 

539 """Escreve o site em `saida` (substituído por inteiro); devolve o nº de páginas.""" 

540 metadados = json.loads((export / "metadados.json").read_text(encoding="utf-8")) 

541 campos = AlertaSite.__dataclass_fields__ 

542 alertas = [ 

543 AlertaSite(**{k: v for k, v in linha.items() if k in campos}) 

544 for linha in _ler_csv(export / ALERTAS.arquivo) 

545 ] 

546 avisos_por_alerta: dict[str, list[AvisoSite]] = defaultdict(list) 

547 for linha in _ler_csv(export / ALERTA_AVISOS.arquivo): 

548 avisos_por_alerta[linha["cod_alerta"]].append( 

549 AvisoSite(linha["evento"], linha["severidade"]) 

550 ) 

551 cobertura = _ler_csv(export / COBERTURA.arquivo) 

552 primeiras = [a["primeira_coleta_utc"] for a in _ler_csv(export / AVISOS.arquivo)] 

553 contexto = Contexto( 

554 atualizado_utc=str(metadados["backup_instante_utc"]), 

555 inicio_avisos_utc=min(primeiras) if primeiras else "", 

556 ) 

557 

558 temporario = preparar_temporario(saida) 

559 (temporario / "municipio").mkdir() 

560 (temporario / "dados").mkdir() 

561 (temporario / "estilo.css").write_text(ESTILO, encoding="utf-8") 

562 (temporario / "index.html").write_text( 

563 gerar_index(alertas, avisos_por_alerta, cobertura, contexto), encoding="utf-8" 

564 ) 

565 (temporario / "dados.html").write_text(gerar_dados(metadados, contexto), encoding="utf-8") 

566 

567 por_municipio: dict[str, list[AlertaSite]] = defaultdict(list) 

568 for a in alertas: 

569 por_municipio[a.codibge].append(a) 

570 for codibge, lista in por_municipio.items(): 

571 (temporario / "municipio" / f"{codibge}.html").write_text( 

572 gerar_municipio(lista, avisos_por_alerta, contexto), encoding="utf-8" 

573 ) 

574 for arquivo in [*(t.arquivo for t in TABELAS), "DICIONARIO.md", "metadados.json"]: 

575 shutil.copyfile(export / arquivo, temporario / "dados" / arquivo) 

576 

577 trocar_diretorio(temporario, saida) 

578 return 2 + len(por_municipio)
