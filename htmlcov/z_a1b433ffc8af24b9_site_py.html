1"""Gera o site estático do acervo a partir de uma exportação (issue #158). 

2 

3Lê só os arquivos de `alertavida.acervo.exportar` — nunca o banco — e escreve 

4HTML + um CSS, sem JavaScript, com links relativos (abre por `file://` e 

5funciona em qualquer subcaminho de hospedagem): 

6 

7- `index.html` — alertas ativos no backup, cobertura dos últimos 30 dias, 

8 municípios por UF; 

9- `municipio/<codibge>.html` — um por município com ao menos um alerta; 

10- `dados.html` + `dados/` — os CSVs, o dicionário e os metadados. 

11 

12Todo texto vindo das fontes passa por `html.escape`. Horários na tela em 

13horário de Brasília (UTC−3, sem horário de verão desde 2019); os CSVs ficam em 

14UTC. Determinístico: a mesma exportação gera os mesmos bytes. 

15 

16Uso: 

17 

18 uv run python -m alertavida.acervo.site \\ 

19 --export /var/lib/alertavida/export --saida /var/lib/alertavida/site 

20"""

123def nome_proprio(nome: str) -> str: 

124 """`JUIZ DE FORA` → `Juiz de Fora`: o CEMADEN publica em maiúsculas.""" 

125 palavras = nome.lower().split() 

126 return " ".join( 

127 p if i > 0 and p in _PARTICULAS else p[:1].upper() + p[1:] for i, p in enumerate(palavras)

193 (a for a in alertas if a.status == "ATIVO"), key=lambda a: (a.inicio_utc, a.cod_alerta)

195 partes = [ 

196 "<h1>Alertas do CEMADEN, com o contexto do INMET</h1>", 

197 _AVISO_OFICIAL, 

198 "<p>Histórico dos alertas publicados pelo CEMADEN, como o coletor do AlertaVida os "

201 f"<h2>Ativos em {em_brasilia(instante_backup)}</h2>", 

202 ] 

203 if ativos: 

204 partes.append( 

205 _tabela( 

206 ["Início", "Município", "Tipo", "Nível", "Avisos do INMET no início"], 

207 ( 

208 [ 

209 em_brasilia(a.inicio_utc), 

210 _link_municipio(a, ""), 

211 escape(a.tipo), 

212 escape(a.nivel_legivel), 

213 _avisos(avisos_por_alerta.get(a.cod_alerta, [])), 

214 ] 

215 for a in ativos 

216 ), 

217 ) 

218 ) 

219 else: 

220 partes.append("<p>Nenhum alerta ativo.</p>") 

221 

222 fim = _instante(instante_backup).date() 

223 recentes = [fim - timedelta(days=d) for d in range(DIAS_RECENTES)] 

224 novos_por_dia: dict[date, int] = defaultdict(int) 

225 for a in alertas: 

226 novos_por_dia[_instante(a.inicio_utc).date()] += 1 

227 cobertura_cemaden = { 

228 c["dia"]: float(c["cobertura"]) for c in cobertura if c["fonte"] == "CEMADEN"

230 partes += [

232 '<p class="suave">Cobertura: fração do dia (em UTC) em que o coletor estava ligado e '

234 _tabela( 

235 ["Dia (UTC)", "Alertas iniciados", "Cobertura da coleta"], 

236 ( 

237 [ 

238 dia.strftime("%d/%m/%Y"), 

239 str(novos_por_dia.get(dia, 0)), 

240 f"{cobertura_cemaden[dia.isoformat()]:.0%}" 

241 if dia.isoformat() in cobertura_cemaden 

242 else '<span class="suave">sem registro</span>', 

243 ] 

244 for dia in recentes 

245 ), 

246 ), 

247 ] 

248 

249 por_uf: dict[str, dict[str, tuple[str, int]]] = defaultdict(dict) 

250 for a in alertas: 

251 nome, total = por_uf[a.uf].get(a.codibge, (a.municipio, 0)) 

252 por_uf[a.uf][a.codibge] = (nome, total + 1) 

253 partes.append(f"<h2>Municípios com alertas ({sum(len(m) for m in por_uf.values())})</h2>")

256 partes.append(f'<h3>{escape(uf)}</h3>\n<ul class="municipios">')

321def gerar_site(export: Path, saida: Path) -> int: 

322 """Escreve o site em `saida` (substituído por inteiro); devolve o nº de páginas.""" 

323 metadados = json.loads((export / "metadados.json").read_text(encoding="utf-8")) 

324 campos = AlertaSite.__dataclass_fields__ 

325 alertas = [ 

326 AlertaSite(**{k: v for k, v in linha.items() if k in campos}) 

327 for linha in _ler_csv(export / ALERTAS.arquivo) 

328 ] 

329 avisos_por_alerta: dict[str, list[AvisoSite]] = defaultdict(list) 

330 for linha in _ler_csv(export / ALERTA_AVISOS.arquivo): 

331 avisos_por_alerta[linha["cod_alerta"]].append( 

332 AvisoSite(linha["evento"], linha["severidade"]) 

333 ) 

334 cobertura = _ler_csv(export / COBERTURA.arquivo) 

335 rodape = ( 

336 f"Gerado a partir de {escape(str(metadados['backup']))} " 

337 f"(SHA-256 {escape(str(metadados['backup_sha256']))[:12]}…). " 

338 "Fontes: CEMADEN (alertas) e INMET (avisos), coletados pelo AlertaVida." 

339 ) 

340 

341 temporario = preparar_temporario(saida) 

342 (temporario / "municipio").mkdir() 

343 (temporario / "dados").mkdir() 

344 (temporario / "estilo.css").write_text(ESTILO, encoding="utf-8") 

345 (temporario / "index.html").write_text( 

346 gerar_index( 

347 alertas, avisos_por_alerta, cobertura, str(metadados["backup_instante_utc"]), rodape 

348 ), 

349 encoding="utf-8", 

350 ) 

351 (temporario / "dados.html").write_text(gerar_dados(metadados, rodape), encoding="utf-8") 

352 

353 por_municipio: dict[str, list[AlertaSite]] = defaultdict(list) 

354 for a in alertas: 

355 por_municipio[a.codibge].append(a) 

356 for codibge, lista in por_municipio.items(): 

357 (temporario / "municipio" / f"{codibge}.html").write_text( 

358 gerar_municipio(lista, avisos_por_alerta, rodape), encoding="utf-8" 

359 ) 

360 for arquivo in [*(t.arquivo for t in TABELAS), "DICIONARIO.md", "metadados.json"]: 

361 shutil.copyfile(export / arquivo, temporario / "dados" / arquivo) 

362 

363 trocar_diretorio(temporario, saida) 

364 return 2 + len(por_municipio)

367def main(argv: list[str] | None = None) -> int: 

368 parser = argparse.ArgumentParser(description="Gera o site estático do acervo AlertaVida.") 

369 parser.add_argument("--export", type=Path, required=True, help="diretório da exportação") 

370 parser.add_argument("--saida", type=Path, required=True, help="diretório do site (substituído)") 

371 args = parser.parse_args(argv) 

372 print(f"{gerar_site(args.export, args.saida)} páginas em {args.saida}") 

373 return 0
