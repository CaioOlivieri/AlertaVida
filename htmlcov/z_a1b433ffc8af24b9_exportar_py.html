1"""Exporta o acervo em CSV a partir de um backup (issue #158). 

2 

3Lê o backup com `mode=ro&immutable=1` — nunca o banco vivo — e escreve num 

4diretório: 

5 

6- `alertas_cemaden.csv`, `avisos_inmet.csv`, `aviso_municipios.csv`, 

7 `alerta_avisos.csv`, `cobertura.csv`; 

8- `DICIONARIO.md`, gerado das MESMAS definições de coluna que os cabeçalhos 

9 dos CSVs (não há como os dois divergirem); 

10- `metadados.json` — de qual backup veio (nome + SHA-256) e contagens. 

11 

12Determinística: nenhum carimbo de geração; o mesmo backup produz os mesmos 

13bytes. A cobertura junta a tabela `rodadas` (#161) às extrações do journal 

14em `wiki/raw/rodadas-journal-*.csv` (antes da tabela existir). 

15 

16Uso: 

17 

18 uv run python -m alertavida.acervo.exportar \\ 

19 --backup /var/lib/alertavida/backups --saida /var/lib/alertavida/export

311def coletas_do_banco(conexao: sqlite3.Connection) -> Iterator[Coleta]: 

312 """Backups anteriores a #161 não têm `rodadas`: a cobertura deles está toda 

313 nas extrações do journal.""" 

314 existe = conexao.execute( 

315 "SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'rodadas'" 

316 ).fetchone() 

317 if existe is None: 

318 return 

319 for iniciada_em, fonte, falha in conexao.execute( 

320 "SELECT iniciada_em, fonte, falha FROM rodadas" 

321 ): 

322 yield Coleta(parse_iso_utc(iniciada_em), fonte, bool(falha))

336def calcular_cobertura(coletas: Iterable[Coleta], ate: datetime) -> Iterator[list[object]]: 

337 """Por (dia, fonte): janelas com ao menos uma coleta bem-sucedida. 

338 

339 O período conhecido de cada fonte vai da sua primeira coleta registrada 

340 (com ou sem falha) até `ate` (o instante do backup). Só janelas que já 

341 TERMINARAM em `ate` contam — uma janela em curso no backup não é falta de 

342 coleta. 

343 """ 

344 por_fonte: dict[str, list[Coleta]] = defaultdict(list) 

345 for coleta in coletas: 

346 if coleta.fonte in JANELA_SEGUNDOS and coleta.instante < ate: 

347 por_fonte[coleta.fonte].append(coleta) 

348 

349 saida: list[list[object]] = [] 

350 for fonte, lista in por_fonte.items(): 

351 janela = JANELA_SEGUNDOS[fonte] 

352 primeira = math.floor(min(c.instante for c in lista).timestamp() / janela) 

353 ultima = math.floor(ate.timestamp() / janela) - 1 

354 if ultima < primeira: 

355 continue 

356 no_periodo = [c for c in lista if math.floor(c.instante.timestamp() / janela) <= ultima] 

357 cobertas = {math.floor(c.instante.timestamp() / janela) for c in no_periodo if not c.falha} 

358 falhas_por_dia: dict[date, int] = defaultdict(int) 

359 for c in no_periodo: 

360 if c.falha: 

361 falhas_por_dia[c.instante.date()] += 1 

362 

363 dia = datetime.fromtimestamp(primeira * janela, UTC).date() 

364 fim = datetime.fromtimestamp(ultima * janela, UTC).date() 

365 while dia <= fim: 

366 inicio_dia = datetime(dia.year, dia.month, dia.day, tzinfo=UTC).timestamp() 

367 de = max(primeira, int(inicio_dia) // janela) 

368 ate_janela = min(ultima, int(inicio_dia + 86400) // janela - 1) 

369 possiveis = ate_janela - de + 1 

370 com_coleta = sum(1 for k in range(de, ate_janela + 1) if k in cobertas) 

371 saida.append( 

372 [ 

373 dia.isoformat(), 

374 fonte, 

375 possiveis, 

376 com_coleta, 

377 f"{com_coleta / possiveis:.4f}", 

378 falhas_por_dia.get(dia, 0), 

379 ] 

380 ) 

381 dia += timedelta(days=1) 

382 saida.sort(key=lambda linha: (str(linha[0]), str(linha[1]))) 

383 yield from saida

386def gerar_dicionario(tabelas: Sequence[Tabela] = TABELAS) -> str: 

387 partes = ["# Dicionário de dados — acervo AlertaVida", "", _AVISO_DICIONARIO]

402def _escrever(destino: Path, tabela: Tabela, linhas: Iterable[Sequence[object]]) -> int: 

403 total = 0 

404 with (destino / tabela.arquivo).open("w", encoding="utf-8", newline="") as saida: 

405 escritor = csv.writer(saida, lineterminator="\n") 

406 escritor.writerow(tabela.cabecalho) 

407 for linha in linhas: 

408 if len(linha) != len(tabela.colunas): 

409 raise ValueError( 

410 f"{tabela.arquivo}: {len(linha)} campos, esperado {len(tabela.colunas)}" 

411 ) 

412 escritor.writerow(linha) 

413 total += 1 

414 return total

417def _sha256(caminho: Path) -> str: 

418 digest = hashlib.sha256() 

419 with caminho.open("rb") as entrada: 

420 for bloco in iter(lambda: entrada.read(1 << 20), b""): 

421 digest.update(bloco) 

422 return digest.hexdigest()

425def preparar_temporario(saida: Path) -> Path: 

426 """Diretório irmão `<saida>.tmp`, vazio — escreve-se nele e troca-se no fim.""" 

427 temporario = saida.with_name(saida.name + ".tmp") 

428 if temporario.exists():

434def trocar_diretorio(temporario: Path, saida: Path) -> None: 

435 """Substitui `saida` por `temporario`: uma execução que falha no meio nunca

437 if saida.exists(): 

438 shutil.rmtree(saida)

442def exportar(backup: Path, saida: Path, rodadas_historicas: Sequence[Path]) -> dict[str, object]: 

443 """Escreve o acervo em `saida` (substituído por inteiro) e devolve os metadados.""" 

444 ate = instante_do_backup(backup) 

445 temporario = preparar_temporario(saida) 

446 

447 with contextlib.closing(abrir_somente_leitura(backup)) as conexao: 

448 contagens = { 

449 ALERTAS.arquivo: _escrever(temporario, ALERTAS, linhas_alertas(conexao)), 

450 AVISOS.arquivo: _escrever(temporario, AVISOS, linhas_avisos(conexao)), 

451 AVISO_MUNICIPIOS.arquivo: _escrever( 

452 temporario, AVISO_MUNICIPIOS, linhas_aviso_municipios(conexao) 

453 ), 

454 ALERTA_AVISOS.arquivo: _escrever( 

455 temporario, ALERTA_AVISOS, linhas_alerta_avisos(conexao) 

456 ), 

457 } 

458 coletas = [*coletas_historicas(rodadas_historicas), *coletas_do_banco(conexao)] 

459 contagens[COBERTURA.arquivo] = _escrever( 

460 temporario, COBERTURA, calcular_cobertura(coletas, ate) 

461 ) 

462 

463 (temporario / "DICIONARIO.md").write_text(gerar_dicionario(), encoding="utf-8") 

464 metadados: dict[str, object] = { 

465 "versao_formato": VERSAO_FORMATO, 

466 "backup": backup.name, 

467 "backup_sha256": _sha256(backup), 

468 "backup_instante_utc": ate.strftime("%Y-%m-%dT%H:%M:%SZ"), 

469 "rodadas_historicas": [p.name for p in rodadas_historicas], 

470 "linhas": contagens, 

471 } 

472 (temporario / "metadados.json").write_text( 

473 json.dumps(metadados, ensure_ascii=False, indent=2, sort_keys=True) + "\n", encoding="utf-8" 

474 ) 

475 

476 trocar_diretorio(temporario, saida) 

477 return metadados
