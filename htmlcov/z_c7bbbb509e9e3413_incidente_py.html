1"""`Incidente` — agregado de N `Alerta`s referindo-se ao mesmo evento físico 

2observado por fontes diferentes (Camada 5, issue #58). 

3 

4Núcleo 100% puro: nenhum I/O, nenhuma consulta a banco. Decidir QUAIS 

5alertas formam um `Incidente` é responsabilidade de `domain/correlacao.py`; 

6este módulo só modela o agregado resultante e a agregação de severidade 

7(Round 1, Q3 — ver wiki/projects/layer-5-correlation.md). 

8 

9Fusão de incidentes (`fundido_em`) e a regra de "quando resolver" dependem de 

10estado persistido (id, membros reais em banco) e ficam para #59/#61 — este 

11módulo modela a forma do agregado, não as transições de ciclo de vida. 

12""" 

13 

14from __future__ import annotations 

15 

16from datetime import datetime 

17from enum import StrEnum 

18from typing import Final 

19 

20from pydantic import BaseModel, ConfigDict, Field, model_validator 

21 

22from alertavida.domain.enums import FonteDado, NivelRisco 

23 

24# Ordem de severidade para a agregação por MÁXIMO (Round 1, Q3). INDETERMINADO 

25# fica de fora deste mapa de propósito: é excluído do máximo, nunca tratado 

26# como "mais baixo que BAIXO" — isso transformaria ausência de informação em 

27# informação, exatamente o erro que a decisão evita. 

28_ORDEM_SEVERIDADE: Final[dict[NivelRisco, int]] = { 

29 NivelRisco.BAIXO: 0, 

30 NivelRisco.MODERADO: 1, 

31 NivelRisco.ALTO: 2, 

32 NivelRisco.MUITO_ALTO: 3, 

33} 

34

36class StatusIncidente(StrEnum): 

37 """Espelha `alertas.status_interno` (Round 1, Q5)."""

39 ATIVO = "ATIVO"

43class MembroIncidente(BaseModel): 

44 """Referência a um `Alerta` membro, pela chave natural (fonte, cod_alerta) 

45 — a mesma que `UNIQUE (fonte, cod_alerta)` garante em `alertas` 

46 (ver [[components/domain-models]]). 

47 

48 Carrega `nivel_risco` além da chave porque a agregação de severidade do 

49 `Incidente` precisa dele e este módulo não faz I/O para buscá-lo — quem 

50 monta o `Incidente` (infra, #59/#61) já tem o `Alerta` em mãos. 

51 """ 

52 

53 fonte: FonteDado 

54 cod_alerta: str = Field(min_length=1) 

55 nivel_risco: NivelRisco 

56 

57 model_config = ConfigDict(frozen=True)

60class Incidente(BaseModel): 

61 """Agregado de `Alerta`s que descrevem o mesmo evento físico. 

62 

63 Espelha o ciclo de vida já existente em `alertas.status_interno` 

64 (Round 1, Q5): ATIVO/RESOLVIDO, deve poder reativar. Resolve só quando 

65 TODOS os membros resolvem — mas essa transição depende do estado real dos 

66 Alertas em banco, então a regra em si vive na integração (#61); aqui só a 

67 forma (`status`/`resolvido_em`) é modelada. 

68 """ 

69 

70 membros: tuple[MembroIncidente, ...] = Field(min_length=1) 

71 status: StatusIncidente = StatusIncidente.ATIVO 

72 criado_em: datetime 

73 atualizado_em: datetime 

74 resolvido_em: datetime | None = None 

75 

76 model_config = ConfigDict(frozen=True) 

77 

78 @model_validator(mode="after") 

79 def _validar_membros_unicos(self) -> "Incidente": 

80 chaves = [(m.fonte, m.cod_alerta) for m in self.membros] 

81 if len(chaves) != len(set(chaves)): 

82 raise ValueError("Incidente com membros duplicados (mesma fonte + cod_alerta)") 

83 return self 

84 

85 @model_validator(mode="after") 

86 def _validar_invariante_resolucao(self) -> "Incidente": 

87 """Invariante atômica, mesmo estilo de 

88 `Alerta._validar_invariante_classificacao`: 

89 resolvido_em IS NOT NULL ⇔ status == RESOLVIDO. 

90 """ 

91 tem_resolvido_em = self.resolvido_em is not None 

92 esta_resolvido = self.status == StatusIncidente.RESOLVIDO 

93 if tem_resolvido_em != esta_resolvido: 

94 raise ValueError( 

95 "Invariante violada: resolvido_em e status devem mudar juntos. " 

96 f"resolvido_em={self.resolvido_em!r}, status={self.status.value}" 

97 ) 

98 return self 

99 

100 @property 

101 def severidade(self) -> NivelRisco | None: 

102 """Severidade agregada = MÁXIMO entre membros com nível conhecido. 

103 

104 Nunca a média — um `INDETERMINADO` não pode diluir um `MUITO_ALTO` 

105 (Round 1, Q3). Retorna `None` quando todos os membros são 

106 `INDETERMINADO` (severidade desconhecida, não "baixa"). 

107 """ 

108 conhecidos = [m.nivel_risco for m in self.membros if m.nivel_risco in _ORDEM_SEVERIDADE] 

109 if not conhecidos: 

110 return None 

111 return max(conhecidos, key=lambda nivel: _ORDEM_SEVERIDADE[nivel]) 

112 

113 @property 

114 def membros_sem_severidade(self) -> int: 

115 """Quantos membros publicam `NivelRisco.INDETERMINADO`. 

116 

117 Contados, mas excluídos de `severidade` — ausência de severidade é 

118 registrada, nunca tratada como evidência de baixa severidade. 

119 """ 

120 return sum(1 for m in self.membros if m.nivel_risco == NivelRisco.INDETERMINADO)
