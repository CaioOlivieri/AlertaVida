1"""Núcleo puro de decisão de correlação — Camada 5 (issue #58). 

2 

3Decide se dois candidatos (tipicamente dois `Alerta`s, ou um `Alerta` e um 

4`Incidente` já aberto — quem chama decide o que cada lado de 

5`CandidatoCorrelacao` representa) descrevem o MESMO evento físico. Segue a 

6arquitetura de record linkage (Fellegi & Sunter) adotada pela camada: 

7blocking → scoring → decisão de três saídas. Este módulo é só o 

8"scoring → decisão"; blocking (índice espacial, janela de tempo) fica em 

9#60. Ver wiki/projects/layer-5-correlation.md — as duas rodadas de 

10clarificação são a spec. 

11 

12Núcleo 100% puro: nenhum I/O, nenhum SQL, nenhum import de `database`/ 

13`sources`. Todos os pesos e limiares abaixo são constantes de módulo, 

14PROVISÓRIAS e conservadoras (viés para separar — Round 2, "false merge vs. 

15false split"). Calibração é escopo da issue #63 (dormente); não ajustar 

16aqui. 

17 

18Cascata causal (ex.: tempestade → enchente → deslizamento) é uma relação 

19DISTINTA de identidade e está fora do escopo deste módulo e da v1 inteira 

20(Round 1, Q4; "v1 scope decisions" #4 na página do projeto). Este módulo só 

21responde "é o mesmo evento?", nunca "um evento desencadeou o outro?". 

22"""

115def distancia_haversine_km(lat1: float, lon1: float, lat2: float, lon2: float) -> float: 

116 """Distância geodésica (grande círculo) entre dois pontos, em km. 

117 

118 Comparações na trilha de decisão NUNCA usam graus decimais puros (Round 

119 1, Q2 — 1° de longitude difere de 1° de latitude em km nas latitudes 

120 brasileiras, a mesma distorção documentada no buffer de 

121 `domain/geographic.py`). Bbox em graus é aceitável só no estágio de 

122 blocking (#60), que nunca decide sozinho. 

123 """ 

124 lat1_rad = math.radians(lat1) 

125 lon1_rad = math.radians(lon1) 

126 lat2_rad = math.radians(lat2) 

127 lon2_rad = math.radians(lon2) 

128 dlat = lat2_rad - lat1_rad 

129 dlon = lon2_rad - lon1_rad 

130 a = ( 

131 math.sin(dlat / 2) ** 2 

132 + math.cos(lat1_rad) * math.cos(lat2_rad) * math.sin(dlon / 2) ** 2 

133 ) 

134 return 2 * _RAIO_TERRA_KM * math.asin(math.sqrt(a))

137# --------------------------------------------------------------------------- 

138# Tabela de compatibilidade de identidade — DIAGONAL-ONLY por construção. 

139# `compatibilidade_tipo` só retorna FORTE/FRACA quando os dois lados caem no 

140# MESMO grupo COBRADE (prefixo) ou no mesmo TipoEvento; nunca entre grupos 

141# diferentes. Um par fora da diagonal (ex.: HIDROLOGICO correlacionando com 

142# METEOROLOGICO — "tempestade causa enchente") é CASCATA CAUSAL, não 

143# identidade (Round 1, Q4), e está fora de escopo. Adicionar uma exceção fora 

144# da diagonal exige evidência empírica documentada — mesma disciplina de 

145# `EVENTO_CEMADEN_PARA_COBRADE` em `domain/cobrade.py`. Não inventar por 

146# suposição. 

147# ---------------------------------------------------------------------------

150def _mesmo_subgrupo_cobrade(a: str, b: str) -> bool: 

151 return a.split(".")[:3] == b.split(".")[:3]

154def _mesmo_grupo_cobrade(a: str, b: str) -> bool: 

155 return a.split(".")[:2] == b.split(".")[:2]

158def compatibilidade_tipo( 

159 cobrade_a: str | None, 

160 tipo_a: TipoEvento, 

161 cobrade_b: str | None, 

162 tipo_b: TipoEvento, 

163) -> ResultadoCompatibilidadeTipo: 

164 """Compara identidade no nível mais específico disponível dos dois lados. 

165 

166 - Ambos com `cobrade_codigo`: compara por prefixo — mesmo subgrupo 

167 (5 níveis, 3 primeiros iguais) = FORTE; mesmo grupo mas subgrupo 

168 diferente (2 primeiros iguais) = FRACA; grupos diferentes = 

169 INCOMPATIVEL. 

170 - Um ou ambos sem `cobrade_codigo`: cai para `TipoEvento` (fallback). 

171 `TipoEvento.INDETERMINADO` em qualquer lado → INDETERMINADA (nunca 

172 vincula sozinho — no máximo REVISAO, decidido em `decidir_correlacao`). 

173 Mesmo `TipoEvento` (nenhum INDETERMINADO) → FORTE. Tipos diferentes → 

174 INCOMPATIVEL. 

175 """ 

176 if cobrade_a is not None and cobrade_b is not None: 

177 if _mesmo_subgrupo_cobrade(cobrade_a, cobrade_b): 

178 return ResultadoCompatibilidadeTipo.FORTE 

179 if _mesmo_grupo_cobrade(cobrade_a, cobrade_b): 

180 return ResultadoCompatibilidadeTipo.FRACA 

181 return ResultadoCompatibilidadeTipo.INCOMPATIVEL 

182 

183 if tipo_a is TipoEvento.INDETERMINADO or tipo_b is TipoEvento.INDETERMINADO: 

184 return ResultadoCompatibilidadeTipo.INDETERMINADA 

185 

186 if tipo_a is tipo_b: 

187 return ResultadoCompatibilidadeTipo.FORTE 

188 

189 return ResultadoCompatibilidadeTipo.INCOMPATIVEL

200def _pontuacao_codigo_ibge(a: int | None, b: int | None) -> float | None: 

201 """`None` = sem evidência (ao menos um lado sem código); não entra no peso.""" 

202 if a is None or b is None: 

203 return None 

204 return 1.0 if a == b else 0.0

207def _pontuar( 

208 a: CandidatoCorrelacao, b: CandidatoCorrelacao 

209) -> tuple[float, ResultadoCompatibilidadeTipo, float, float]: 

210 """Média ponderada sobre a evidência DISPONÍVEL — pesos de evidência 

211 ausente (tipicamente `codigo_ibge` em pares CEMADEN×EONET) são 

212 redistribuídos entre o resto, nunca tratados como pontuação zero. 

213 """ 

214 componentes: list[tuple[float, float]] = [] 

215 

216 pontuacao_ibge = _pontuacao_codigo_ibge(a.codigo_ibge, b.codigo_ibge) 

217 if pontuacao_ibge is not None: 

218 componentes.append((PESO_CODIGO_IBGE, pontuacao_ibge)) 

219 

220 distancia_km = distancia_haversine_km(a.latitude, a.longitude, b.latitude, b.longitude) 

221 pontuacao_distancia = max(0.0, 1.0 - distancia_km / DISTANCIA_MAXIMA_KM) 

222 componentes.append((PESO_DISTANCIA, pontuacao_distancia)) 

223 

224 delta_t_segundos = (b.momento_onset - a.momento_onset).total_seconds() 

225 pontuacao_tempo = max(0.0, 1.0 - abs(delta_t_segundos) / JANELA_TEMPO_SEGUNDOS) 

226 componentes.append((PESO_TEMPO, pontuacao_tempo)) 

227 

228 compatibilidade = compatibilidade_tipo( 

229 a.cobrade_codigo, a.tipo_evento, b.cobrade_codigo, b.tipo_evento 

230 ) 

231 componentes.append((PESO_TIPO, _PONTUACAO_POR_COMPATIBILIDADE[compatibilidade])) 

232 

233 peso_total = sum(peso for peso, _ in componentes) 

234 score = sum(peso * pontuacao for peso, pontuacao in componentes) / peso_total 

235 return score, compatibilidade, distancia_km, delta_t_segundos

238def decidir_correlacao(a: CandidatoCorrelacao, b: CandidatoCorrelacao) -> DecisaoCorrelacao: 

239 """Decide VINCULA / NAO_VINCULA / REVISAO para um par de candidatos. 

240 

241 Dois portões estruturais valem ANTES de qualquer limiar de score — não 

242 dependem de calibração e não podem ser contornados por reajuste de pesos 

243 (#63): tipos INCOMPATIVEL nunca vinculam, e `TipoEvento.INDETERMINADO` 

244 nunca vincula sozinho (no máximo REVISAO, quando o resto da evidência é 

245 forte). 

246 """ 

247 score, compatibilidade, distancia_km, delta_t_segundos = _pontuar(a, b) 

248 

249 if compatibilidade is ResultadoCompatibilidadeTipo.INCOMPATIVEL: 

250 return DecisaoCorrelacao( 

251 resultado=ResultadoDecisao.NAO_VINCULA, 

252 score=score, 

253 motivo="tipos_incompativeis", 

254 distancia_km=distancia_km, 

255 delta_t_segundos=delta_t_segundos, 

256 ) 

257 

258 if compatibilidade is ResultadoCompatibilidadeTipo.INDETERMINADA: 

259 if score >= LIMIAR_REVISAO: 

260 resultado = ResultadoDecisao.REVISAO 

261 motivo = "tipo_indeterminado_evidencia_forte" 

262 else: 

263 resultado = ResultadoDecisao.NAO_VINCULA 

264 motivo = "tipo_indeterminado_evidencia_fraca" 

265 return DecisaoCorrelacao( 

266 resultado=resultado, 

267 score=score, 

268 motivo=motivo, 

269 distancia_km=distancia_km, 

270 delta_t_segundos=delta_t_segundos, 

271 ) 

272 

273 if score >= LIMIAR_VINCULA: 

274 resultado = ResultadoDecisao.VINCULA 

275 motivo = "evidencia_forte" 

276 elif score >= LIMIAR_REVISAO: 

277 resultado = ResultadoDecisao.REVISAO 

278 motivo = "evidencia_ambigua" 

279 else: 

280 resultado = ResultadoDecisao.NAO_VINCULA 

281 motivo = "evidencia_fraca" 

282 

283 return DecisaoCorrelacao( 

284 resultado=resultado, 

285 score=score, 

286 motivo=motivo, 

287 distancia_km=distancia_km, 

288 delta_t_segundos=delta_t_segundos, 

289 )
